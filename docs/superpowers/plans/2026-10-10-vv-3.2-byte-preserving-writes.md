# vv 3.2 — Byte-Preserving Writes Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A vv write never changes a byte outside the span it edits, `board` accepts the flag spelling and never silently matches nothing, and the adoption report fails closed when its denominator is dark — shipped as 3.2.0.

**Architecture:** Both engines keep their existing `split("\n")` line model (elements carry their own trailing `\r`). The fix deletes the one whole-file re-termination loop in `splice` and replaces every whole-file `eol_of(text)` decision with a local rule: inserted lines inherit the terminator of the nearest terminated neighbour, `patch` aligns the unchanged head and tail of a section by common prefix/suffix so a read→patch round-trip is a byte no-op, and frontmatter edits replace only the edited key's line. A new standalone suite pins expected bytes from literal tables (independent of both engines), and the CI fixture vault gains mixed-ending notes so CI can see the class.

**Tech Stack:** Python 3 (`src/vv_impl.py`, semantic authority), Rust (`vrust/src/write.rs`, native fast path; `cargo fmt`/`clippy -D warnings`), standalone `tests/test_*.py` scripts with their own `check()`/exit-1 convention, `run_tests.sh` as the gate.

**Spec:** `docs/superpowers/specs/2026-10-10-vv-3.2-byte-preserving-writes-and-adoption-design.md`

```python
def sha8_of(engine, vault, note, sid):
    """The sha8 the engine prints for a section — what a caller pastes into patch."""
    r = subprocess.run(engine + ["outline", note], capture_output=True, text=True,
                       env=dict(os.environ, VV_VAULT=vault, VV_NO_METRICS="1"))
    for line in r.stdout.splitlines():
        cols = line.split("\t")
        if cols and cols[0] == sid:
            return cols[4]
    raise AssertionError(f"no {sid} in outline: {r.stdout!r} {r.stderr!r}")


def run_case(engine, name, fixture, argv, stdin, expected):
    vault = tempfile.mkdtemp(prefix="vv-eol-")
    try:
        os.makedirs(os.path.join(vault, "Standups"))
        note = "N.md"
        with open(os.path.join(vault, note), "wb") as f:
            f.write(fixture)
        argv = list(argv)
        if argv[0] == "patch":
            argv[2] = sha8_of(engine, vault, note, argv[1])
        cmd = engine + [argv[0], note] + argv[1:]
        r = subprocess.run(cmd, input=stdin, capture_output=True,
                           env=dict(os.environ, VV_VAULT=vault, VV_NO_METRICS="1"))
        with open(os.path.join(vault, note), "rb") as f:
            got = f.read()
        ok = r.returncode == 0 and got == expected
        return ok, f"exit={r.returncode} stderr={r.stderr[:200]!r}\n   got={got!r}\n   exp={expected!r}"
    finally:
        shutil.rmtree(vault, ignore_errors=True)

    if os.path.exists(VR[0]):

            if not ok:

    if fails:
        print(f"\n{len(fails)} of {n} FAILED: {fails}")
        return 1
    print(f"ALL PASS (eol-preserve: {n})")
    return 0
```

```python
def _elem_cr(lines, i):
    """Does element i of a split("\\n") list end its line with CRLF? The final
    element is unterminated by construction, so it never counts."""
    return i < len(lines) - 1 and lines[i].endswith("\r")

def _insert_cr(lines, at):
    """Terminator style for lines inserted before index `at`: the nearest
    terminated line above, else the nearest below, else LF."""
    last_term = len(lines) - 1            # elements with index < last_term are terminated
    above = min(at, last_term) - 1
    if above >= 0:
        return lines[above].endswith("\r")
    if at < last_term:
        return lines[at].endswith("\r")
    return False

def _last_eol(text):
    """The terminator of the last terminated line in `text`, else LF."""
    i = text.rfind("\n")
    if i < 0:
        return "\n"
    return "\r\n" if i > 0 and text[i - 1] == "\r" else "\n"

def _fm_parts(text):
    """Frontmatter as terminator-carrying pieces, or None when there is none:
    bom, eol1 (after the opening ---), fm_lines (split on \\n, elements keep
    their \\r), eol2 (before the closing ---), tail (after the closing ---),
    body. _fm_join(*parts) == text for any note that has frontmatter."""
    fm, body, tail, bom = split_fm_full(text)
    if fm is None:
        return None
    head = text[:len(text) - len(body)]          # bom + --- + eol1 + fm + eol2 + --- + tail
    after_open = head[len(bom) + 3:]
    eol1 = "\r\n" if after_open.startswith("\r\n") else "\n"
    rest = after_open[len(eol1) + len(fm):]      # eol2 + --- + tail
    eol2 = rest[:len(rest) - 3 - len(tail)]
    return bom, eol1, fm.split("\n"), eol2, tail, body

def _fm_join(bom, eol1, fm_lines, eol2, tail, body):
    return bom + "---" + eol1 + "\n".join(fm_lines) + eol2 + "---" + tail + body
```

```python
def splice(lines, start, end, new_lines):
    """Replace lines[start:end] with `new_lines` (bare content, no line terminators)
    and return the file text.

    Bytes outside the replaced span are copied. Inside it, lines that match the
    old span's head or tail (compared without their \\r) keep their original
    bytes; the changed middle takes the terminator of the first changed original
    line, or of the nearest terminated neighbour when the span was empty. The
    file's EOF-newline property is preserved exactly as it was. `lines` is a
    split on "\\n", so a CRLF line carries a trailing "\\r" and a file ending in
    a newline has "" as its last element."""
    ended_with_newline = bool(lines) and lines[-1] == ""
    old = list(lines[start:end])
    body = [b.rstrip("\r") for b in new_lines]
    if end >= len(lines):
        # the replaced span reaches EOF: restore the file's original EOF-newline property
        if ended_with_newline:
            if not body or body[-1] != "":
                body.append("")
        elif body and body[-1] == "":
            body.pop()
    old_c = [o.rstrip("\r") for o in old]
    p = 0
    while p < len(old_c) and p < len(body) and old_c[p] == body[p]:
        p += 1
    s = 0
    while s < len(old_c) - p and s < len(body) - p and old_c[-1 - s] == body[-1 - s]:
        s += 1
    head = old[:p]
    tail = old[len(old) - s:] if s else []
    middle = body[p:len(body) - s]
    if p < len(old_c) - s:
        cr = _elem_cr(lines, start + p)          # first changed original line
    else:
        cr = _insert_cr(lines, start + p)        # pure insertion
    middle = [m + "\r" if cr else m for m in middle]
    merged = list(lines[:start]) + head + middle + tail + list(lines[end:])
    if merged and merged[-1].endswith("\r") and not (lines and lines[-1].endswith("\r")):
        merged[-1] = merged[-1][:-1]             # the final element is unterminated; we must not invent a \r there
    return "\n".join(merged)
```

```rust
fn elem_cr(lines: &[&str], i: usize) -> bool {
    // the final element is unterminated by construction, so it never counts
    i + 1 < lines.len() && lines[i].ends_with('\r')
}

fn insert_cr(lines: &[&str], at: usize) -> bool {
    // nearest terminated line above the insertion point, else below, else LF
    let last_term = lines.len().saturating_sub(1); // elements with index < last_term are terminated
    let above = at.min(last_term);
    if above > 0 {
        return lines[above - 1].ends_with('\r');
    }
    if at < last_term {
        return lines[at].ends_with('\r');
    }
    false
}

fn last_eol(text: &str) -> &'static str {
    match text.rfind('\n') {
        None => "\n",
        Some(i) => {
            if i > 0 && text.as_bytes()[i - 1] == b'\r' {
                "\r\n"
            } else {
                "\n"
            }
        }
    }
}

    let fm = fm?;
    let head = &text[..text.len() - body.len()];

        bom: bom.to_string(),

        eol2: eol2.to_string(),
        tail: tail.to_string(),
        body: body.to_string(),
    })
}

        p.eol1,
        p.fm_lines.join("\n"),
        p.eol2,
        p.tail,
```

```rust
    let end_c = end.min(lines.len());

            body.pop();

        }
```
